1//! A utility module for managing key debounce logic.
2//!
3//! This module provides the `KeyDebounce` struct, which facilitates detecting
4//! freshly pressed keys in a key input system. It tracks the state of keys
5//! and determines if a key has been transitioned to a pressed state (`true`)
6//! from a non-pressed state (`false`).
7
8use crate::input::InputIntent;
9
10/// A utility struct for managing key debounce logic.
11///
12/// `KeyDebounce` tracks the press state of keys and provides a method to check
13/// if a key has been freshly pressed (transitioned from not being pressed to pressed).
14pub struct KeyDebounce {
15    /// A hashmap that stores the last known state (`true` for pressed, `false` for not pressed)
16    /// of each key.
17    last: std::collections::HashMap<InputIntent, bool>,
18}
19
20impl KeyDebounce {
21    /// Creates a new instance of `KeyDebounce`.
22    ///
23    /// # Returns
24    ///
25    /// A new `KeyDebounce` instance with an empty key state map.
26    pub fn new() -> Self {
27        Self {
28            last: std::collections::HashMap::new(),
29        }
30    }
31
32    /// Checks if a key has been freshly pressed.
33    ///
34    /// A keypress is considered fresh if it transitioned from a non-pressed
35    /// state (`false`) to a pressed state (`true`). This method also updates
36    /// the internal state of the key.
37    ///
38    /// # Parameters
39    ///
40    /// - `key`: The key to check.
41    /// - `is_down`: The current state of the key (`true` if pressed, `false` otherwise).
42    ///
43    /// # Returns
44    ///
45    /// Returns `true` if the key was freshly pressed, otherwise `false`.
46    pub fn fresh_press_detected(&mut self, intent: InputIntent, is_down: bool) -> bool {
47        let was_down = self.last.get(&intent).copied().unwrap_or(false);
48        self.last.insert(intent, is_down);
49        is_down && !was_down
50    }
51}
