14pub fn start_logic_thread(
15    tx_state: Sender<GameState>,
16    tx_main: Sender<MainEvent>,
17    input_event_rx: Receiver<InputEvent>,
18) {
19    thread::spawn(move || {
20        let mut state = GameState::default();
21
22        let mut next_frame_due = Instant::now();
23
24        let sleeper = spin_sleep::SpinSleeper::new(0)
25            .with_spin_strategy(spin_sleep::SpinStrategy::YieldThread);
26
27        loop {
28            while let Ok(event) = input_event_rx.try_recv() {
29                match event {
30                    InputEvent::ToggleCRT => state.crt = !state.crt,
31                    InputEvent::EscapePressed => {
32                        tx_main.send(MainEvent::Shutdown).ok();
33                    }
34                    InputEvent::IncreasePressed => {
35                        if let SceneBackground::Plasma(plasma) = &mut state.background {
36                            plasma.intensity = (plasma.intensity + 0.1).clamp(0.0, 1.0);
37                            debug!("Plasma intensity: {:.2}", plasma.intensity);
38                        }
39                    }
40                    InputEvent::DecreasePressed => {
41                        if let SceneBackground::Plasma(plasma) = &mut state.background {
42                            plasma.intensity = (plasma.intensity - 0.1).clamp(0.0, 1.0);
43                            debug!("Plasma intensity: {:.2}", plasma.intensity);
44                        }
45                    }
46                    InputEvent::NumberSelected(num) => match num {
47                        0 => state.background = SceneBackground::None,
48                        1 => state.background = SceneBackground::Plasma(Default::default()),
49                        2 => state.background = SceneBackground::Fire,
50                        3 => state.background = SceneBackground::Matrix,
51                        _ => (),
52                    },
53                    InputEvent::ToggleItalic => {
54                        state.italic = !state.italic;
55                    }
56                }
57            }
58
59            state.tick += 1;
60            state.x = (state.x + 1) % WIDTH;
61            // state.y = (state.y + 1) % HEIGHT;
62            // state.y = HEIGHT / 2;
63            state.y = ((SIN_LUT[(state.tick & 0xFF) as usize] as usize * HEIGHT / 768)
64                + HEIGHT / 2)
65                .clamp(0, HEIGHT);
66
67            if tx_state.send(state.clone()).is_err() {
68                warn!("Failed to send game state update");
69            }
70
71            // Schedule next frame time exactly `LOGIC_DURATION` after previous
72            next_frame_due += LOGIC_DURATION;
73
74            let now = Instant::now();
75            if now < next_frame_due {
76                sleeper.sleep_until(next_frame_due);
77            } else {
78                warn!(
79                    "Logic frame missed deadline by {:?}",
80                    now.duration_since(next_frame_due)
81                );
82                next_frame_due = now;
83            }
84        }
85    });
86}
