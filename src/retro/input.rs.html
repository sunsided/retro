1//! This module provides key handling utilities for defining and managing key bindings, debouncing key inputs,
2//! and processing high-level key intents.
3//!
4//! It contains the following submodules:
5//! - `bindings`: Used to define specific key mappings and their associated high-level intents.
6//! - `debounce`: Facilitates key debounce handling to avoid unintended repeated input.
7//! - `handling`: Provides mechanisms to handle key input events and map them to high-level actions.
8//!
9//! The module makes use of the `minifb` crate for key input handling and incorporates pre-defined
10//! key bindings for common actions, such as toggling a CRT effect, modifying settings, or exiting a context.
11
12mod bindings;
13mod debounce;
14mod handling;
15
16use bindings::define_key_bindings;
17pub use debounce::KeyDebounce;
18pub use handling::KeyHandling;
19use minifb::Key;
20
21// Defines specific key bindings and their corresponding high-level intents.
22define_key_bindings! {
23    /// Exit or escape the current context.
24    Escape, escape => Key::Escape,
25    /// Toggle the CRT effect.
26    ToggleCrt, toggle_crt => Key::Space,
27    /// Increase a specific setting.
28    Increase, increase => Key::Up,
29    /// Decrease a specific setting.
30    Decrease, decrease => Key::Down,
31    /// Switch to setting zero
32    Number0, num_0 => Key::Key0,
33    /// Switch to setting one
34    Number1, num_1 => Key::Key1,
35    /// Switch to setting two
36    Number2, num_2 => Key::Key2,
37    /// Switch to setting three
38    Number3, num_3 => Key::Key3,
39    /// Toggle italic fonts
40    Italic, italic => Key::I,
41}
42
43#[derive(Debug, Clone, Copy)]
44pub enum InputEvent {
45    ToggleCRT,
46    EscapePressed,
47    IncreasePressed,
48    DecreasePressed,
49    NumberSelected(usize),
50    ToggleItalic,
51}
