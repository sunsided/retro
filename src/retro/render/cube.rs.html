9/// Rotates a 3D vector around the Y-axis by a specified angle.
10///
11/// # Parameters
12/// - `v`: The vector to rotate.
13/// - `angle`: The angle to rotate by, in radians.
14///
15/// # Returns
16/// The rotated vector.
17pub fn rotate_y(v: Vec3, angle: f32) -> Vec3 {
18    let (sin, cos) = angle.sin_cos();
19    Vec3(v.0 * cos + v.2 * sin, v.1, -v.0 * sin + v.2 * cos)
20}

22/// Projects a 3D vector onto the 2D screen using a simple perspective transformation.
23///
24/// # Parameters
25/// - `v`: The 3D vector to project.
26///
27/// # Returns
28/// A tuple containing the projected 2D screen coordinates as `(x, y)`.
29pub fn project(v: Vec3) -> (usize, usize) {
30    let scale = 100.0 / (v.2 + 3.0); // Simple perspective
31    let x = (v.0 * scale + WIDTH as f32 / 2.0).round() as isize;
32    let y = (v.1 * scale + HEIGHT as f32 / 2.0).round() as isize;
33    (
34        x.clamp(0, (WIDTH - 1) as isize) as usize,
35        y.clamp(0, (HEIGHT - 1) as isize) as usize,
36    )
37}

39/// Draws a rotating wireframe cube onto a framebuffer.
40///
41/// # Parameters
42/// - `buffer`: A mutable reference to the framebuffer where the cube will be drawn.
43/// - `frame`: The current frame count, which determines the rotation angle.
44/// - `color`: The color of the cube's edges.
45///
46/// # Functionality
47/// This function creates a cube represented by 8 vertices in 3D space. It rotates
48/// these vertices around the Y-axis based on the `frame` parameter, projects them
49/// onto a 2D screen, and renders the edges of the cube by connecting the appropriate
50/// vertices using the `draw_line` function.
51pub fn draw_wireframe_cube(buffer: &mut FrameBuffer, frame: usize, color: Color) {
52    let angle = (frame as f32) * PI / 180.0;
53
54    let vertices = [
55        Vec3(-1.0, -1.0, -1.0),
56        Vec3(1.0, -1.0, -1.0),
57        Vec3(1.0, 1.0, -1.0),
58        Vec3(-1.0, 1.0, -1.0),
59        Vec3(-1.0, -1.0, 1.0),
60        Vec3(1.0, -1.0, 1.0),
61        Vec3(1.0, 1.0, 1.0),
62        Vec3(-1.0, 1.0, 1.0),
63    ];
64
65    let edges = [
66        (0, 1),
67        (1, 2),
68        (2, 3),
69        (3, 0), // back face
70        (4, 5),
71        (5, 6),
72        (6, 7),
73        (7, 4), // front face
74        (0, 4),
75        (1, 5),
76        (2, 6),
77        (3, 7), // connections
78    ];
79
80    let transformed: Vec<(usize, usize)> = vertices
81        .iter()
82        .map(|&v| project(rotate_y(v, angle)))
83        .collect();
84
85    for &(i, j) in &edges {
86        let (x0, y0) = transformed[i];
87        let (x1, y1) = transformed[j];
88        draw_line(buffer, x0, y0, x1, y1, color);
89    }
90}
