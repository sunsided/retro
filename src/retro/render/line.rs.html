3/// Draws a line on a framebuffer using Bresenham's line algorithm.
4///
5/// # Parameters
6/// - `buffer`: A mutable reference to the framebuffer where the line will be drawn.
7/// - `x0`, `y0`: The starting point of the line (coordinates in pixels).
8/// - `x1`, `y1`: The ending point of the line (coordinates in pixels).
9/// - `color`: The color of the line, represented as a 32-bit integer.
10///
11/// # Functionality
12/// This function calculates a line between the `(x0, y0)` and `(x1, y1)` coordinates
13/// using Bresenham's line-drawing algorithm. It then modifies the provided framebuffer
14/// by setting the pixels along the line to the specified color. The framebuffer is assumed
15/// to be a 1-dimensional array of pixels with a width of `WIDTH` and height of `HEIGHT`.
16///
17/// # Important Notes
18/// - The function clamps drawing to the bounds of the framebuffer (`WIDTH` by `HEIGHT`);
19///   pixels outside these dimensions are ignored.
20/// - Ensure that the framebuffer has been properly initialized before calling this function.
21///
22/// # Example
23/// ```rust
24/// use retro::render::line::draw_line;
25/// use retro::{HEIGHT, WIDTH};
26///
27/// let mut buffer = [0; WIDTH * HEIGHT];
28/// draw_line(&mut buffer, 10, 10, 100, 50, 0xFF00FF); // Draw a purple line
29/// ```
30pub const fn draw_line(
31    buffer: &mut FrameBuffer,
32    x0: usize,
33    y0: usize,
34    x1: usize,
35    y1: usize,
36    color: u32,
37) {
38    let mut x0 = x0 as isize;
39    let mut y0 = y0 as isize;
40    let x1 = x1 as isize;
41    let y1 = y1 as isize;
42
43    let dx = (x1 - x0).abs();
44    let dy = -(y1 - y0).abs();
45    let sx = if x0 < x1 { 1 } else { -1 };
46    let sy = if y0 < y1 { 1 } else { -1 };
47    let mut err = dx + dy;
48
49    loop {
50        if x0 >= 0 && y0 >= 0 && x0 < WIDTH as isize && y0 < HEIGHT as isize {
51            buffer[y0 as usize * WIDTH + x0 as usize] = color;
52        }
53        if x0 == x1 && y0 == y1 {
54            break;
55        }
56        let e2 = 2 * err;
57        if e2 >= dy {
58            err += dy;
59            x0 += sx;
60        }
61        if e2 <= dx {
62            err += dx;
63            y0 += sy;
64        }
65    }
66}
