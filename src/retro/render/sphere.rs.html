1use crate::render::cube::{Vec3, project, rotate_y};
2use crate::render::line::draw_line;
3use crate::{Color, FrameBuffer};
4use std::f32::consts::PI;
5
6/// Draws a rotating wireframe sphere onto the framebuffer.
7///
8/// # Arguments
9///
10/// * `buffer` - A mutable reference to the framebuffer where the sphere will be rendered.
11/// * `frame` - The current frame number, used to calculate the rotation angle of the sphere.
12/// * `color` - The color of the sphere's wireframe.
13pub fn draw_wireframe_sphere(buffer: &mut FrameBuffer, frame: usize, color: Color) {
14    let angle = (frame as f32) * PI / 180.0;
15    let lat_segments = 12; // Number of latitude segments (horizontal slices of the sphere).
16    let lon_segments = 24; // Number of longitude segments (vertical slices of the sphere).
17    let radius = 1.0;
18
19    // Iterate through latitude segments to create the sphere's grid.
20    for i in 0..lat_segments {
21        // Angles for the current and next latitude segment.
22        let theta1 = PI * (i as f32) / (lat_segments as f32);
23        let theta2 = PI * ((i + 1) as f32) / (lat_segments as f32);
24
25        // Iterate through longitude segments to create the sphere's grid.
26        for j in 0..lon_segments {
27            // Angles for the current and next longitude segment.
28            let phi1 = 2.0 * PI * (j as f32) / (lon_segments as f32);
29            let phi2 = 2.0 * PI * ((j + 1) as f32) / (lon_segments as f32);
30
31            // Calculate the 3D points for the current grid on the sphere.
32            let p1 = Vec3(
33                radius * theta1.sin() * phi1.cos(),
34                radius * theta1.cos(),
35                radius * theta1.sin() * phi1.sin(),
36            );
37            let p2 = Vec3(
38                radius * theta1.sin() * phi2.cos(),
39                radius * theta1.cos(),
40                radius * theta1.sin() * phi2.sin(),
41            );
42            let p3 = Vec3(
43                radius * theta2.sin() * phi1.cos(),
44                radius * theta2.cos(),
45                radius * theta2.sin() * phi1.sin(),
46            );
47
48            // Rotate and project the 3D points onto the 2D screen.
49            let (x1, y1) = project(rotate_y(p1, angle));
50            let (x2, y2) = project(rotate_y(p2, angle));
51            let (x3, y3) = project(rotate_y(p3, angle));
52
53            // Draw lines connecting the points on the 2D screen to form the sphere's wireframe.
54            draw_line(buffer, x1, y1, x2, y2, color);
55            draw_line(buffer, x1, y1, x3, y3, color);
56        }
57    }
58}
