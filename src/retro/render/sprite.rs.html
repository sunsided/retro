19/// Draws a sprite onto the framebuffer with optional scaling.
20///
21/// # Parameters
22/// - `buffer`: A mutable reference to the framebuffer where the sprite will be drawn.
23/// - `sprite`: A slice containing the sprite's color indices.
24/// - `sprite_width`: The width of the sprite in pixels.
25/// - `x`: The X-coordinate to begin drawing the sprite on the screen.
26/// - `y`: The Y-coordinate to begin drawing the sprite on the screen.
27/// - `scale`: The scaling factor to apply to the sprite. E.g., a factor of 2 doubles its size.
28///
29/// # Behavior
30/// - Pixels with a color index of `0` in the sprite are treated as transparent and skipped.
31/// - Colors are mapped from the sprite's indices to the `palettes::STANDARD` palette.
32/// - The function automatically handles clipping if parts of the sprite extend beyond the screen boundaries.
33pub fn draw_sprite(
34    buffer: &mut FrameBuffer,
35    sprite: &[PaletteIndex; 8 * 8],
36    sprite_width: usize,
37    x: usize,
38    y: usize,
39    scale: usize,
40) {
41    let sprite_height = sprite.len() / sprite_width;
42    for sy in 0..sprite_height {
43        for sx in 0..sprite_width {
44            let color_index = sprite[sy * sprite_width + sx];
45            if color_index == 0 {
46                continue; // Transparent
47            }
48            for dy in 0..scale {
49                for dx in 0..scale {
50                    let px = x.wrapping_add(sx * scale + dx); // Compute pixel X position
51                    let py = y.wrapping_add(sy * scale + dy); // Compute pixel Y position
52                    if px < WIDTH && py < HEIGHT {
53                        buffer[py * WIDTH + px] = palettes::STANDARD[color_index as usize];
54                    }
55                }
56            }
57        }
58    }
59}
