15    pub fn update(&mut self) {
16        for x in 0..WIDTH {
17            self.heat[(HEIGHT - 1) * WIDTH + x] = if fastrand::u8(0..10) > 1 { 255 } else { 0 };
18        }
19
20        for y in 1..HEIGHT {
21            for x in 0..WIDTH {
22                let src = y * WIDTH + x;
23                let below = src + WIDTH;
24                let decay = fastrand::u32(1..20);
25                let new_val = self
26                    .heat
27                    .get(below)
28                    .copied()
29                    .unwrap_or(0)
30                    .saturating_sub(decay);
31                let dx = fastrand::usize(0..3);
32                let dst_x = x.wrapping_add(dx).wrapping_sub(1).min(WIDTH - 1);
33                let dst_y = y - 1;
34                let dst = dst_y * WIDTH + dst_x;
35                self.heat[dst] = new_val;
36            }
37        }
38    }
39
40    pub fn render(&self, buffer: &mut FrameBuffer) {
41        for (i, &val) in self.heat.iter().enumerate() {
42            buffer[i] = palettes::FIRE[val as usize];
43        }
