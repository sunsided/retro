1//! This module defines the [`GlyphIndex`] struct, which represents the index
2//! of a glyph within an alphabet. The struct provides utility methods for
3//! constructing and accessing the internal index, while maintaining type
4//! safety. Additionally, the module implements `PartialEq` to allow direct
5//! comparison between a `GlyphIndex` and a `u8` value.
6
7/// The index of a glyph within an alphabet.
8#[derive(Debug, Clone, Copy, PartialEq, Eq, Ord, PartialOrd)]
9pub struct GlyphIndex(u8);
10
11impl GlyphIndex {
12    pub const fn new(index: u8) -> Self {
13        Self(index)
14    }
15
16    pub const fn into_inner(self) -> u8 {
17        self.0
18    }
19}
20
21impl PartialEq<u8> for GlyphIndex {
22    fn eq(&self, other: &u8) -> bool {
23        self.0 == *other
24    }
25}
