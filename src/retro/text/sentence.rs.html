1//! This module provides structures and utilities for working with font glyphs
2//! and sentences composed of these glyphs. It includes:
3//!
4//! - The `Sentence` struct: Represents multiple [`Glyphs`] glued together,
5//!   allowing efficient manipulation and iteration of text data.
6//! - The `Glyphs` struct: Represents a slice of glyph references from the
7//!   same alphabet, providing convenient access to the underlying data.
8//! - The `font8x8_sentence` macro: Facilitates the creation of `Sentence`
9//!   objects from literals, enabling a concise and readable syntax for
10//!   embedding text data with specified alphabets.
11//!
12//! The main use case of this module is to manage and render text efficiently,
13//! especially in graphics or text-intensive applications where control over
14//! individual glyphs and alphabets is necessary.
15
16use std::borrow::Cow;
17use std::ops::Deref;
18
19macro_rules! font8x8_sentence {
20    (
21        $(
22            $( $alphabet:path : )? $text:literal
23        )+
24        $(,)?
25    ) => {{
26        const GLYPHS_ARR: &[$crate::text::sentence::Glyphs] = &[
27            $(
28                {
29                    const ALPHABET: $crate::text::Alphabet = font8x8_sentence!(@resolve_alphabet $( $alphabet )?);
30
31                    const GLYPHS: [$crate::text::GlyphIndex; $text.len()] = {
32                        let mut out = [$crate::text::GlyphIndex::new(0); $text.len()];
33                        let bytes = $text.as_bytes();
34                        let mut i = 0;
35                        while i < $text.len() {
36                            let ch = bytes[i] as char;
37                            out[i] = ALPHABET.map_char_index_or_panic(ch);
38                            i += 1;
39                        }
40                        out
41                    };
42                    $crate::text::sentence::Glyphs::new(ALPHABET, std::borrow::Cow::Borrowed(&GLYPHS))
43                }
44            ),+
45        ];
46        $crate::text::Sentence::new(GLYPHS_ARR)
47    }};
48
49    (@resolve_alphabet) => {
50        $crate::text::Alphabet::Basic
51    };
52    (@resolve_alphabet $alphabet:path) => {
53        $alphabet
54    };
55}
56
57macro_rules! font8x8_sentence_runtime {
58    (
59        $( $alphabet:path : $text:expr ),+ $(,)?
60    ) => {{
61        let mut glyphs_vec = Vec::new();
62        $(
63            {
64                let glyphs: Vec<$crate::text::GlyphIndex> = $text
65                    .chars()
66                    .map(|ch| $alphabet.map_char_index_or_panic(ch))
67                    .collect();
68                glyphs_vec.push($crate::text::sentence::Glyphs::new($alphabet, std::borrow::Cow::Owned(glyphs)));
69            }
70        )+
71        $crate::text::Sentence::new(Box::leak(glyphs_vec.into_boxed_slice()))
72    }};
73
74    (
75        $( $text:expr ),+ $(,)?
76    ) => {{
77        let mut glyphs_vec = Vec::new();
78        $(
79            {
80                let alphabet = $crate::text::Alphabet::Basic;
81                let glyphs: Vec<$crate::text::GlyphIndex> = $text
82                    .chars()
83                    .map(|ch| alphabet.map_char_index_or_panic(ch))
84                    .collect();
85                glyphs_vec.push($crate::text::sentence::Glyphs::new(alphabet, std::borrow::Cow::Owned(glyphs)));
86            }
87        )+
88        $crate::text::Sentence::new(Box::leak(glyphs_vec.into_boxed_slice()))
89    }};
90}

97/// Multiple [`Glyphs`] glued together.
98#[derive(Debug, Clone)]
99pub struct Sentence<'a>(&'a [Glyphs<'a>]);
100
101impl<'a> Sentence<'a> {
102    pub const fn new(glyphs: &'a [Glyphs<'a>]) -> Self {
103        Self(glyphs)
104    }
105
106    pub fn iter(&self) -> std::slice::Iter<Glyphs<'a>> {
107        self.0.iter()
108    }
109}
110
111/// A slice of glyph references from the same alphabet.
112#[derive(Debug, Clone)]
113pub struct Glyphs<'a>(Alphabet, Cow<'a, [GlyphIndex]>);
114
115impl<'a> Glyphs<'a> {
116    pub const fn new(alphabet: Alphabet, glyphs: Cow<'a, [GlyphIndex]>) -> Self {
117        Self(alphabet, glyphs)
118    }
119
120    pub const fn alphabet(&self) -> Alphabet {
121        self.0
122    }
123
124    pub fn bytes(&'a self) -> std::slice::Iter<'a, GlyphIndex> {
125        self.1.iter()
126    }
127}
128
129impl PartialEq<&[u8]> for Glyphs<'_> {
130    fn eq(&self, other: &&[u8]) -> bool {
131        self.1.as_ref().eq(*other)
132    }
133}
134
135impl<const N: usize> PartialEq<[u8; N]> for Glyphs<'_> {
136    fn eq(&self, other: &[u8; N]) -> bool {
137        self.1.as_ref().eq(other)
138    }
139}
140
141impl Deref for Glyphs<'_> {
142    type Target = [GlyphIndex];
143
144    fn deref(&self) -> &Self::Target {
145        self.1.as_ref()
146    }
147}

149#[cfg(test)]
150mod tests {
151    use crate::text::{Alphabet, GlyphIndex, font8x8_sentence};
152
153    #[test]
154    fn test_font8x8_sentence() {
155        let sentence = font8x8_sentence!(
156            "Hello"
157            Alphabet::Basic: "World",
158        );
159
160        let mut iter = sentence.iter();
161
162        // Test first Glyphs: "Hello"
163        let first_glyphs = iter.next().expect("Expected first glyphs");
164        assert_eq!(first_glyphs.alphabet(), Alphabet::Basic);
165        let hello_indices: Vec<GlyphIndex> = first_glyphs.bytes().cloned().collect();
166        assert_eq!(hello_indices.len(), 5); // "Hello" has 5 characters
167
168        // Test second Glyphs: "World"
169        let second_glyphs = iter.next().expect("Expected second glyphs");
170        assert_eq!(second_glyphs.alphabet(), Alphabet::Basic);
171        let world_indices: Vec<GlyphIndex> = second_glyphs.bytes().cloned().collect();
172        assert_eq!(world_indices.len(), 5); // "World" has 5 characters
173
174        // Ensure there are no more Glyphs
175        assert!(iter.next().is_none());
176    }
177}
