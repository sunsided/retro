11pub const fn make_fire_palette() -> [u32; 256] {
12    let mut palette = [0u32; 256];
13    let mut i = 0;
14    while i < 256 {
15        let (r, g, b) = if i <= 63 {
16            (i * 4, 0, 0) // Black → Red
17        } else if i <= 127 {
18            (255, (i - 64) * 4, 0) // Red → Orange
19        } else if i <= 191 {
20            (255, 255, (i - 128) * 4) // Orange → Yellow
21        } else {
22            (255, 255, 255) // Yellow → White
23        };
24        palette[i] = ((r as u32) << 16) | ((g as u32) << 8) | b as u32;
25        i += 1;
26    }
27    palette
28}
29
30pub fn make_plasma_palette() -> [u32; 256] {
31    let mut palette = [0u32; 256];
32    let mut i = 0;
33    while i < 256 {
34        let angle = i as f32 * std::f32::consts::PI * 2.0 / 256.0;
35        let r = ((angle.sin() * 127.0 + 128.0) as u8).saturating_sub(32);
36        let g = (angle.cos() * 127.0 + 128.0) as u8;
37        let b = (angle.sin().cos() * 127.0 + 128.0) as u8;
38        palette[i] = ((r as u32) << 16) | ((g as u32) << 8) | b as u32;
39        i += 1;
40    }
41    palette
42}

44macro_rules! apply_hex_to_u32 {
45    ($arr:expr) => {{
46        let mut result = [0u32; $arr.len()];
47        let mut i = 0;
48        while i < $arr.len() {
49            result[i] = $crate::colors::hex_to_u32($arr[i]);
50            i += 1;
51        }
52        result
53    }};
54}
