29enum MainEvent {
30    Shutdown,
31}

33pub fn run_game() {
34    let (input_event_tx, input_event_rx) = channel::unbounded::<InputEvent>();
35    let (tx_state, rx_state) = channel::bounded::<GameState>(1);
36    let (tx_main, rx_main) = channel::unbounded::<MainEvent>();
37    let (ready_tx, ready_rx) = channel::bounded::<Box<FrameBuffer>>(NUM_BUFFERS);
38    let (free_tx, free_rx) = channel::bounded::<Box<FrameBuffer>>(NUM_BUFFERS);
39
40    for _ in 0..NUM_BUFFERS {
41        let buffer = Box::new([Color::default(); WIDTH * HEIGHT]);
42        free_tx.send(buffer).unwrap();
43    }
44
45    start_render_thread(rx_state, ready_tx, free_rx);
46    start_logic_thread(tx_state, tx_main, input_event_rx);
47
48    let mut window = Window::new(
49        "Plasma Animation - ESC to exit",
50        WIDTH,
51        HEIGHT,
52        WindowOptions {
53            resize: false,
54            scale: minifb::Scale::X4,
55            ..WindowOptions::default()
56        },
57    )
58    .unwrap();
59
60    let bindings = KeyBindings::default();
61    let mut input_handling = KeyHandling::new(bindings, input_event_tx.clone());
62
63    while window.is_open() {
64        input_handling.process_input(&window).ok();
65
66        select! {
67            recv(ready_rx) -> msg => match msg {
68                Ok(buffer_arc) => {
69                    window
70                        .update_with_buffer(buffer_arc.as_slice(), WIDTH, HEIGHT)
71                        .unwrap();
72                    free_tx.send(buffer_arc).unwrap();
73                }
74                Err(_) => {
75                    eprintln!("Error receiving buffer event");
76                    break;
77                },
78            },
79
80            recv(rx_main) -> msg => match msg {
81                Ok(event) => {
82                    match event {
83                        MainEvent::Shutdown => break,
84                    }
85                }
86                Err(_) => {
87                    eprintln!("Error receiving output event");
88                    break;
89                },
90            },
91
92            default(INPUT_DURATION) => {
93                // Neither frame nor event received in INPUT_DURATION
94            }
95        }
96    }
97}
