use crate::map::tile::Tile;
use crate::map::tile_coord::ChunkLocalTileCoord;

/// A [`TileChunk`] is a densely-packed are of [`Tile`] instances.
#[derive(Debug, Clone)]
pub struct TileChunk {
    pub tiles: [Tile; TileChunk::TILE_COUNT],
}

impl TileChunk {
    /// The number of tiles this chunk is wide.
    pub const WIDTH_IN_TILES: usize = 16;
    /// The number of tiles this chunk is high.
    pub const HEIGHT_IN_TILES: usize = 16;
    /// The total number of tiles in this chunk, i.e. its area.
    pub const TILE_COUNT: usize = Self::WIDTH_IN_TILES * Self::HEIGHT_IN_TILES;

    /// Creates a new chunk pre-filled with empty tiles.
    #[inline]
    pub const fn new_empty() -> Self {
        Self {
            tiles: [Tile::new(); TileChunk::TILE_COUNT],
        }
    }

    #[inline]
    pub fn get(&self, coord: ChunkLocalTileCoord) -> &Tile {
        &self.tiles[coord.to_dense_tile_chunk_index()]
    }

    #[inline]
    pub fn set(&mut self, coord: ChunkLocalTileCoord, tile: Tile) {
        self.tiles[coord.to_dense_tile_chunk_index()] = tile;
    }

    #[inline]
    pub fn iter(&self) -> TileIter {
        self.tiles.iter()
    }

    #[inline]
    pub fn iter_mut(&mut self) -> TileIterMut {
        self.tiles.iter_mut()
    }

    #[inline]
    pub fn iter_with_coords(&self) -> TileWithCoordIter<'_> {
        self.tiles.iter().enumerate().map(|(i, tile)| {
            let coord = ChunkLocalTileCoord::from_dense_tile_chunk_index(i);
            (coord, tile)
        })
    }

    #[inline]
    pub fn iter_mut_with_coords(&mut self) -> TileWithCoordIterMut<'_> {
        self.tiles.iter_mut().enumerate().map(|(i, tile)| {
            let coord = ChunkLocalTileCoord::from_dense_tile_chunk_index(i);
            (coord, tile)
        })
    }
}

pub type TileIter<'a> = core::slice::Iter<'a, Tile>;

pub type TileIterMut<'a> = core::slice::IterMut<'a, Tile>;

pub type TileWithCoordIter<'a> = core::iter::Map<
    core::iter::Enumerate<core::slice::Iter<'a, Tile>>,
    fn((usize, &Tile)) -> (ChunkLocalTileCoord, &Tile),
>;

pub type TileWithCoordIterMut<'a> = core::iter::Map<
    core::iter::Enumerate<core::slice::IterMut<'a, Tile>>,
    fn((usize, &mut Tile)) -> (ChunkLocalTileCoord, &mut Tile),
>;

/// The local coordinates of a [`Tile`](super::tile::Tile) within a [`DenseTileChunk`](TileChunk).
#[derive(Debug, Default, Copy, Clone, Eq, PartialEq, Hash)]
#[repr(C)]
pub struct ChunkLocalTileCoord {
    pub x: u8,
    pub y: u8,
}

/// The local coordinate of a [`DenseTileChunk`](TileChunk) in a [`WorldRegion`].
#[derive(Debug, Default, Copy, Clone, Eq, PartialEq, Hash)]
#[repr(C)]
pub struct RegionLocalChunkCoord {
    pub x: u8,
    pub y: u8,
}

/// The translated local coordinate of a [`Tile`](super::tile::Tile) in a [`WorldRegion`].
#[derive(Debug, Default, Copy, Clone, Eq, PartialEq, Hash)]
#[repr(C)]
pub struct RegionLocalTileCoord {
    pub x: u16,
    pub y: u16,
}

/// The local coordinate of a [`WorldRegion`] within the world.
#[derive(Debug, Default, Copy, Clone, Eq, PartialEq, Hash)]
#[repr(C)]
pub struct WorldGlobalRegionCoord {
    pub x: i8,
    pub y: i8,
}

/// The translated local coordinate of a [`TileChunk`] within the world.
/// Marks the global coordinate offset of the chunk's base coordinate in world space.
///
/// This is typically the outcome of adding region offset and chunk offset. For construction
/// of the map, you're likely more interested in [`WorldGlobalRegionCoord`].
#[derive(Debug, Default, Copy, Clone, Eq, PartialEq, Hash)]
#[repr(C)]
pub struct WorldGlobalChunkCoord {
    pub x: i32,
    pub y: i32,
}

/// The translated local coordinate of a [`Tile`](super::tile::Tile) within the world.
/// Marks the global coordinate the tile coordinate in world space.
///
/// This is typically the outcome of adding region offset, chunk offset and tile offset.
#[derive(Debug, Default, Copy, Clone, Eq, PartialEq, Hash)]
#[repr(C)]
pub struct WorldGlobalTileCoord {
    pub x: i32,
    pub y: i32,
}

impl Add<ChunkLocalTileCoord> for RegionLocalChunkCoord {
    type Output = RegionLocalTileCoord;

    fn add(self, rhs: ChunkLocalTileCoord) -> Self::Output {
        RegionLocalTileCoord {
            x: self.x as u16 * TileChunk::WIDTH_IN_TILES as u16 + rhs.x as u16,
            y: self.y as u16 * TileChunk::HEIGHT_IN_TILES as u16 + rhs.y as u16,
        }
    }
}

impl Add<RegionLocalTileCoord> for WorldGlobalRegionCoord {
    type Output = WorldGlobalTileCoord;

    fn add(self, rhs: RegionLocalTileCoord) -> Self::Output {
        WorldGlobalTileCoord {
            x: self.x as i32 * WorldRegion::WIDTH_IN_TILES as i32 + rhs.x as i32,
            y: self.y as i32 * WorldRegion::HEIGHT_IN_TILES as i32 + rhs.y as i32,
        }
    }
}

impl Add<RegionLocalChunkCoord> for WorldGlobalRegionCoord {
    type Output = WorldGlobalChunkCoord;

    fn add(self, rhs: RegionLocalChunkCoord) -> Self::Output {
        WorldGlobalChunkCoord {
            x: self.x as i32 * WorldRegion::WIDTH_IN_TILES as i32
                + rhs.x as i32 * TileChunk::WIDTH_IN_TILES as i32,
            y: self.y as i32 * WorldRegion::HEIGHT_IN_TILES as i32
                + rhs.y as i32 * TileChunk::HEIGHT_IN_TILES as i32,
        }
    }
}

impl Add<ChunkLocalTileCoord> for WorldGlobalChunkCoord {
    type Output = WorldGlobalTileCoord;

    fn add(self, rhs: ChunkLocalTileCoord) -> Self::Output {
        WorldGlobalTileCoord {
            x: self.x + rhs.x as i32,
            y: self.y + rhs.y as i32,
        }
    }
}

impl ChunkLocalTileCoord {
    #[inline(always)]
    pub const fn new(x: u8, y: u8) -> Self {
        debug_assert!(x < TileChunk::WIDTH_IN_TILES as u8);
        debug_assert!(y < TileChunk::HEIGHT_IN_TILES as u8);
        Self { x, y }
    }

    #[inline(always)]
    pub const fn to_dense_tile_chunk_index(self) -> usize {
        (self.y as usize) * TileChunk::WIDTH_IN_TILES + (self.x as usize)
    }

    #[inline(always)]
    pub const fn from_dense_tile_chunk_index(index: usize) -> Self {
        debug_assert!(index < TileChunk::TILE_COUNT);
        let x = (index % TileChunk::WIDTH_IN_TILES) as u8;
        let y = (index / TileChunk::WIDTH_IN_TILES) as u8;
        Self { x, y }
    }
}

impl RegionLocalChunkCoord {
    #[inline(always)]
    pub const fn new(x: u8, y: u8) -> Self {
        debug_assert!(x < WorldRegion::WIDTH_IN_CHUNKS as u8);
        debug_assert!(y < WorldRegion::HEIGHT_IN_CHUNKS as u8);
        Self { x, y }
    }

    #[inline(always)]
    pub const fn to_dense_region_index(self) -> usize {
        (self.y as usize) * WorldRegion::WIDTH_IN_CHUNKS + (self.x as usize)
    }

    #[inline(always)]
    pub const fn from_dense_region_index(index: usize) -> Self {
        debug_assert!(index < WorldRegion::CHUNK_COUNT);
        let x = (index % WorldRegion::WIDTH_IN_CHUNKS) as u8;
        let y = (index / WorldRegion::WIDTH_IN_CHUNKS) as u8;
        Self { x, y }
    }
}

impl WorldGlobalRegionCoord {
    #[inline(always)]
    pub const fn new(x: i8, y: i8) -> Self {
        Self { x, y }
    }
}

            for x in 0..TileChunk::WIDTH_IN_TILES {
                let original = ChunkLocalTileCoord::new(x as u8, y as u8);

use crate::map::tile::Tile;
use std::collections::HashMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct TileCoord2D {
    pub x: u16,
    pub y: u16,
}

pub type TileCoordX = u16;
pub type TileCoordY = u16;
pub type TileCoordZ = u8;

#[derive(Debug, Clone)]
#[deprecated(note = "Replaced by TileWorld")]
pub struct SparseTileMap {
    pub base: HashMap<TileCoord2D, Tile>,
    pub stacks: HashMap<TileCoord2D, HashMap<TileCoordZ, Tile>>, // (x, y) → z → tile
}

impl SparseTileMap {
    pub fn new() -> Self {
        Self {
            base: HashMap::new(),
            stacks: HashMap::new(),
        }
    }

    pub fn set(&mut self, x: TileCoordX, y: TileCoordY, z: TileCoordZ, tile: Tile) {
        if z == 0 {
            self.base.insert(TileCoord2D { x, y }, tile);
        } else {
            self.stacks
                .entry(TileCoord2D { x, y })
                .or_default()
                .insert(z, tile);
        }
    }

    pub fn get(&self, x: u16, y: u16, z: u8) -> Option<Tile> {
        if z == 0 {
            self.base.get(&TileCoord2D { x, y }).copied()
        } else {
            self.stacks
                .get(&TileCoord2D { x, y })
                .and_then(|map| map.get(&z))
                .copied()
        }
    }

    pub fn all_instances(&self) -> impl Iterator<Item = (u16, u16, u8, Tile)> + '_ {
        let base_iter = self
            .base
            .iter()
            .map(|(&TileCoord2D { x, y }, &tile)| (x, y, 0, tile));
        let stack_iter = self
            .stacks
            .iter()
            .flat_map(|(&TileCoord2D { x, y }, levels)| {
                levels.iter().map(move |(&z, &tile)| (x, y, z, tile))
            });
        base_iter.chain(stack_iter)
    }
}

        if self.wall_south() {
            f(WallDirection::South);
        }
        if self.wall_east() {
            f(WallDirection::East);

        if self.wall_west() {
            f(WallDirection::West);
        }
    }

    pub fn rotation_y_radians(&self) -> f32 {
        use std::f32::consts::FRAC_PI_2;
        match self {
            WallDirection::North => std::f32::consts::PI,
            WallDirection::East => FRAC_PI_2,
            WallDirection::South => 0.0,
            WallDirection::West => FRAC_PI_2 * 3.0,
        }

        match self {
            WallDirection::North => (0, -1),
            WallDirection::East => (1, 0),
            WallDirection::South => (0, 1),
            WallDirection::West => (-1, 0),
        }
    }

        for (wy, wx) in (-1..1).cartesian_product(-1..1) {
            // For each region, create a chunk.
            let mut region = WorldRegion::empty();
            for (cy, cx) in (0..WorldRegion::HEIGHT_IN_CHUNKS)
                .cartesian_product(0..WorldRegion::HEIGHT_IN_CHUNKS)
            {
                // For each chunk, create flat floors.
                let mut chunk = TileChunk::new_empty();
                for (ty, tx) in
                    (0..TileChunk::HEIGHT_IN_TILES).cartesian_product(0..TileChunk::HEIGHT_IN_TILES)
                {
                    let tile = Tile::new()
                        // TODO: Fix when tile atlas is ready
                        .with_floor(FloorType::Empty);

                    // TODO: Create a construct-from-iterator function.
                    chunk.set(ChunkLocalTileCoord::new(tx as u8, ty as u8), tile);
                }

                // TODO: Create a construct-from-iterator function.
                region.set(RegionLocalChunkCoord::new(cx as u8, cy as u8), chunk);
            }
            world.insert(WorldGlobalRegionCoord::new(wx as i8, wy as i8), region);
        }

        world
    }
}

use crate::map::tile_chunk::TileChunk;
use crate::map::tile_coord::RegionLocalChunkCoord;
use std::mem::MaybeUninit;

/// A [`WorldRegion`] is a densely-packed are of [`TileChunk`] instances.
#[derive(Debug, Clone)]
pub struct WorldRegion {
    pub chunks: [TileChunk; WorldRegion::CHUNK_COUNT],
}

impl WorldRegion {
    /// The width of a region in terms of chunks.
    pub const WIDTH_IN_CHUNKS: usize = 2;
    /// The height of a region in terms of chunks.
    pub const HEIGHT_IN_CHUNKS: usize = 2;
    /// The total number of chunks in a region.
    pub const CHUNK_COUNT: usize = Self::WIDTH_IN_CHUNKS * Self::HEIGHT_IN_CHUNKS;

    /// The width of a region in terms of tiles,
    /// based on [`WIDTH_IN_CHUNKS`](Self::WIDTH_IN_CHUNKS) and [`TileChunk::WIDTH_IN_TILES`](TileChunk::WIDTH_IN_TILES).
    pub const WIDTH_IN_TILES: usize = Self::WIDTH_IN_CHUNKS * TileChunk::WIDTH_IN_TILES;
    /// The height of a region in terms of tiles,
    /// based on [`HEIGHT_IN_CHUNKS`](Self::HEIGHT_IN_CHUNKS) and [`TileChunk::HEIGHT_IN_TILES`](TileChunk::HEIGHT_IN_TILES).
    pub const HEIGHT_IN_TILES: usize = Self::HEIGHT_IN_CHUNKS * TileChunk::HEIGHT_IN_TILES;
    /// The total number of tiles in a region.
    pub const TILE_COUNT: usize = Self::WIDTH_IN_TILES * Self::HEIGHT_IN_TILES;

    /// Creates a new chunk pre-filled with empty chunks.
    #[inline]
    pub const fn empty() -> Self {
        let mut chunks: [MaybeUninit<TileChunk>; Self::CHUNK_COUNT] =
            [const { MaybeUninit::uninit() }; Self::CHUNK_COUNT];
        let mut i = 0;
        while i < Self::CHUNK_COUNT {
            chunks[i] = MaybeUninit::new(TileChunk::new_empty());
            i += 1;
        }
        // SAFETY: All entries were just initialized above
        Self {
            chunks: unsafe { std::mem::transmute::<_, [TileChunk; Self::CHUNK_COUNT]>(chunks) },
        }
    }

    #[inline]
    pub fn get(&self, coord: RegionLocalChunkCoord) -> &TileChunk {
        &self.chunks[coord.to_dense_region_index()]
    }

    #[inline]
    pub fn get_mut(&mut self, coord: RegionLocalChunkCoord) -> &mut TileChunk {
        &mut self.chunks[coord.to_dense_region_index()]
    }

    #[inline]
    pub fn set(&mut self, coord: RegionLocalChunkCoord, chunk: TileChunk) {
        self.chunks[coord.to_dense_region_index()] = chunk;
    }

    #[inline]
    pub fn iter(&self) -> ChunkIter {
        self.chunks.iter()
    }

    #[inline]
    pub fn iter_mut(&mut self) -> ChunkIterMut {
        self.chunks.iter_mut()
    }

    #[inline]
    pub fn iter_with_coords(&self) -> ChunkWithCoordIter<'_> {
        self.chunks.iter().enumerate().map(|(i, chunk)| {
            let coord = RegionLocalChunkCoord::from_dense_region_index(i);
            (coord, chunk)
        })
    }

    #[inline]
    pub fn iter_mut_with_coords(&mut self) -> ChunkWithCoordIterMut<'_> {
        self.chunks.iter_mut().enumerate().map(|(i, chunk)| {
            let coord = RegionLocalChunkCoord::from_dense_region_index(i);
            (coord, chunk)
        })
    }
}

pub type ChunkIter<'a> = core::slice::Iter<'a, TileChunk>;

pub type ChunkIterMut<'a> = core::slice::IterMut<'a, TileChunk>;

pub type ChunkWithCoordIter<'a> = core::iter::Map<
    core::iter::Enumerate<core::slice::Iter<'a, TileChunk>>,
    fn((usize, &TileChunk)) -> (RegionLocalChunkCoord, &TileChunk),
>;

pub type ChunkWithCoordIterMut<'a> = core::iter::Map<
    core::iter::Enumerate<core::slice::IterMut<'a, TileChunk>>,
    fn((usize, &mut TileChunk)) -> (RegionLocalChunkCoord, &mut TileChunk),
>;
